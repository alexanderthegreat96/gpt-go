# Python dependencies for the helper scripts in this directory.
#
# These are NOT needed to build, train or run anything in Go. They exist only
# for the two scripts that talk to PyTorch:
#
#   dump_gpt2_reference.py          reference activations, for verify-gpt2
#   dump_gpt2_tokenizer_fixtures.py test fixtures for the byte level BPE
#
# Both are run once per model and the Go side never calls them again. The
# fixtures they produce are committed, so bpe tests pass without Python.
#
# Install:
#
#   pip install -r scripts/requirements.txt
#
# On a machine without an NVIDIA GPU, install torch from the CPU index first
# to avoid pulling in several gigabytes of CUDA you will never use:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install -r scripts/requirements.txt
#
# Versions are lower bounds, not pins. Known working: torch 2.14.1+cpu,
# transformers 5.20.1, safetensors 0.8.0.

# the reference implementation we compare our forward pass against
torch>=2.0

# GPT2LMHeadModel and AutoTokenizer, for the reference dump and fixtures
transformers>=4.30

# reading and writing the .safetensors files both sides exchange
safetensors>=0.3
